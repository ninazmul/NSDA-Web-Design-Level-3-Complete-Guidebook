Unit 9: Work with Core JavaScript

Unit Code: OU-ICT-WD-06-L3-V2
This unit covers JavaScript programming fundamentals for web interactivity, including variables, data types, operators, control structures, functions, DOM manipulation, and event handling.

📋 Overview

JavaScript is the programming language of the web. It adds interactivity, dynamic content, and behavior to your HTML pages. Every modern website uses JavaScript — from simple form validation to complex single-page applications.

💡 Where JavaScript Runs

In the Browser: Every modern browser has a built-in JavaScript engine (V8 in Chrome, SpiderMonkey in Firefox). Use the browser console (F12 → Console) to test code instantly.

On the Server: Node.js allows JavaScript to run on servers, enabling full-stack development with one language.

Adding JavaScript to HTML

HTML
<!-- Method 1: Internal (in HTML file) -->
<script>
  console.log("Hello from JavaScript!");
</script>

<!-- Method 2: External file (recommended) -->
<script src="js/main.js"></script>

<!-- Method 3: Inline (avoid in production) -->
<button onclick="alert('Clicked!')">Click Me</button>

⚠️ == vs === (Very Important!)

== (loose equality) converts types before comparing: 5 == "5" is true

=== (strict equality) compares both value AND type: 5 === "5" is false

Always use === to avoid unexpected bugs!

📚 Arrays

JavaScript
// Creating arrays
const fruits = ["Apple", "Banana", "Cherry"];
const numbers = [1, 2, 3, 4, 5];

// Accessing elements (0-indexed)
fruits[0];          // "Apple"
fruits[fruits.length - 1]; // "Cherry" (last item)

// Common array methods
fruits.push("Date");     // Add to end → ["Apple","Banana","Cherry","Date"]
fruits.pop();             // Remove from end → returns "Date"
fruits.unshift("Avocado");// Add to start
fruits.shift();           // Remove from start
fruits.includes("Banana");// true
fruits.indexOf("Banana"); // 1
fruits.splice(1, 1);     // Remove 1 item at index 1
fruits.slice(0, 2);      // Get items from index 0 to 1

// Iteration methods
fruits.forEach(function(fruit) {
  console.log(fruit);
});

// Map — create new array from transformation
const upperFruits = fruits.map(f => f.toUpperCase());

// Filter — create new array from condition
const longNames = fruits.filter(f => f.length > 5);

// Find — get first match
const found = fruits.find(f => f.startsWith("B"));

🔀 Element 2: Apply Control Structures

❓ Conditional Statements

JavaScript
// if...else if...else
const score = 85;

if (score >= 90) {
  console.log("Grade: A (Excellent)");
} else if (score >= 80) {
  console.log("Grade: B (Very Good)");
} else if (score >= 70) {
  console.log("Grade: C (Good)");
} else if (score >= 60) {
  console.log("Grade: D (Satisfactory)");
} else {
  console.log("Grade: F (Needs Improvement)");
}

// switch statement
const day = "Monday";

switch (day) {
  case "Monday":
  case "Tuesday":
  case "Wednesday":
  case "Thursday":
  case "Friday":
    console.log("Weekday — Time to work!");
    break;
  case "Saturday":
  case "Sunday":
    console.log("Weekend — Time to rest!");
    break;
  default:
    console.log("Invalid day");
}

// Ternary operator (short if/else)
const status = score >= 60 ? "Pass" : "Fail";

🔄 Loops

JavaScript
// for loop — when you know the count
for (let i = 0; i < 5; i++) {
  console.log(`Iteration ${i}`);
}

// for...of — iterate over array values
const colors = ["red", "green", "blue"];
for (const color of colors) {
  console.log(color);
}

// for...in — iterate over object keys
const user = { name: "Nazmul", age: 25, role: "Developer" };
for (const key in user) {
  console.log(`${key}: ${user[key]}`);
}

// while loop — when condition is checked first
let count = 0;
while (count < 3) {
  console.log(`Count: ${count}`);
  count++;
}

// do...while — runs at least once
let num = 10;
do {
  console.log(`Number: ${num}`);
  num++;
} while (num < 5);
// Prints "Number: 10" (runs once even though 10 < 5 is false)

// break and continue
for (let i = 0; i < 10; i++) {
  if (i === 3) continue; // Skip 3
  if (i === 7) break;    // Stop at 7
  console.log(i);         // 0, 1, 2, 4, 5, 6
}

⚙️ Element 3: Use Functions

🔧 Function Types

JavaScript
// 1. Function Declaration
function greet(name) {
  return `Hello, ${name}!`;
}
console.log(greet("Nazmul")); // "Hello, Nazmul!"

// 2. Function Expression
const add = function(a, b) {
  return a + b;
};
console.log(add(5, 3)); // 8

// 3. Arrow Function (ES6 — modern & concise)
const multiply = (a, b) => a * b;
console.log(multiply(4, 5)); // 20

// Arrow function with body
const calculateArea = (width, height) => {
  const area = width * height;
  return area;
};

// 4. Default Parameters
function createUser(name, role = "Student") {
  return { name, role };
}
console.log(createUser("Ali"));           // { name: "Ali", role: "Student" }
console.log(createUser("Ali", "Teacher")); // { name: "Ali", role: "Teacher" }

// 5. Rest Parameters (...args)
function sum(...numbers) {
  return numbers.reduce((total, num) => total + num, 0);
}
console.log(sum(1, 2, 3, 4, 5)); // 15

🔒 Scope and Hoisting

JavaScript
// Global Scope — accessible everywhere
const globalVar = "I'm global";

function example() {
  // Function Scope — only inside this function
  const functionVar = "I'm function-scoped";

  if (true) {
    // Block Scope — only inside this block
    let blockVar = "I'm block-scoped";
    const alsoBlock = "Me too";
    var notBlock = "I escape blocks! (function-scoped)";
  }

  console.log(notBlock);   // ✅ Works (var is function-scoped)
  // console.log(blockVar); // ❌ Error (let is block-scoped)
}

// Hoisting — var declarations are moved to the top
console.log(x); // undefined (hoisted but not initialized)
var x = 5;

// console.log(y); // ❌ ReferenceError (let/const are in TDZ)

🖱️ Element 4: DOM Manipulation and Event Handling

📌 What is the DOM?

The Document Object Model (DOM) is a tree-like representation of your HTML document. JavaScript uses the DOM to read, modify, add, and remove HTML elements dynamically — making web pages interactive without reloading.

🎯 DOM Selection Methods

JavaScript
// Select by ID (returns single element)
const header = document.getElementById("header");

// Select by CSS selector (returns first match)
const firstCard = document.querySelector(".card");

// Select all matches (returns NodeList)
const allCards = document.querySelectorAll(".card");

// Select by class name (returns HTMLCollection)
const buttons = document.getElementsByClassName("btn");

// Select by tag name
const paragraphs = document.getElementsByTagName("p");

🔧 DOM Manipulation

JavaScript

⚡ Event Handling

JavaScript
const button = document.querySelector("#myButton");

// addEventListener — Best practice!
button.addEventListener("click", function(event) {
  console.log("Button clicked!");
  console.log(event.target); // The element that was clicked
});

// Arrow function version
button.addEventListener("click", (e) => {
  e.target.textContent = "Clicked!";
});

// Remove event listener
function handleClick() {
  console.log("Clicked!");
}
button.addEventListener("click", handleClick);
button.removeEventListener("click", handleClick);

💡 Practical Examples

Example 1: Dark Mode Toggle

JavaScript
const toggleBtn = document.querySelector("#theme-toggle");
const body = document.body;

toggleBtn.addEventListener("click", () => {
  body.classList.toggle("dark-mode");

  // Save preference to localStorage
  const isDark = body.classList.contains("dark-mode");
  localStorage.setItem("theme", isDark ? "dark" : "light");

  // Update button text
  toggleBtn.textContent = isDark ? "☀️" : "🌙";
});

// Load saved theme on page load
if (localStorage.getItem("theme") === "dark") {
  body.classList.add("dark-mode");
}

Example 2: Form Validation

JavaScript
const form = document.querySelector("#contact-form");
const emailInput = document.querySelector("#email");
const errorMsg = document.querySelector("#error-message");

form.addEventListener("submit", (e) => {
  e.preventDefault(); // Stop form from submitting

  const email = emailInput.value.trim();

  if (!email) {
    errorMsg.textContent = "Email is required!";
    emailInput.classList.add("error");
    return;
  }

  if (!email.includes("@")) {
    errorMsg.textContent = "Please enter a valid email!";
    emailInput.classList.add("error");
    return;
  }

  // Success
  errorMsg.textContent = "";
  emailInput.classList.remove("error");
  console.log("Form submitted successfully!");
});

Example 3: Dynamic Content (To-Do List)

JavaScript
const input = document.querySelector("#todo-input");
const addBtn = document.querySelector("#add-btn");
const list = document.querySelector("#todo-list");

addBtn.addEventListener("click", () => {
  const text = input.value.trim();
  if (!text) return;

  // Create new list item
  const li = document.createElement("li");
  li.textContent = text;

  // Add delete button
  const deleteBtn = document.createElement("button");
  deleteBtn.textContent = "❌";
  deleteBtn.addEventListener("click", () => li.remove());

  li.appendChild(deleteBtn);
  list.appendChild(li);
  input.value = ""; // Clear input
});
